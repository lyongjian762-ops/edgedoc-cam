__RUN_MODEL_SCRIPT__
#!/bin/sh
# 在板卡本机启动 CPU 推理服务；按 Ctrl+C 停止。
set -eu
BASE="$HOME/ai/xiaomi-ocr-0"
exec "$BASE/llama.cpp/build/bin/llama-server" \
    --host 127.0.0.1 --port 8000 --alias xiaomi-ocr-0 \
    --model "$BASE/models/Xiaomi-OCR-0.Q4_K_M.gguf" \
    --mmproj "$BASE/models/Xiaomi-OCR-0.mmproj-bf16.gguf" \
    --threads 4 --ctx-size 2048 --batch-size 256 --ubatch-size 128 \
    --parallel 1 --n-gpu-layers 0 --jinja
